## Database Schema: Lawyer Workflows

```
CREATE TABLE workflows (
  id SERIAL PRIMARY KEY,
  workflow_key VARCHAR(100), -- 'case_management', 'legal_research', etc.
  workflow_name VARCHAR(255),
  description TEXT,
  category VARCHAR(50), -- 'core', 'optional', 'free', 'paid'
  persona VARCHAR(50), -- 'lawyer', 'paralegal', 'investigator', etc.
  practice_areas TEXT[], -- array of practice areas
  jurisdictions TEXT[], -- array of jurisdictions
  is_free BOOLEAN,
  provider VARCHAR(100), -- 'transcend', 'clio', 'mycase', etc.
  priority_rank INT, -- display order
  created_at TIMESTAMP
);

CREATE TABLE workflow_tools (
  id SERIAL PRIMARY KEY,
  workflow_id INT REFERENCES workflows,
  tool_name VARCHAR(255),
  tool_description TEXT,
  is_free BOOLEAN,
  provider VARCHAR(100),
  url VARCHAR(255), -- for external tools
  integration_type VARCHAR(50), -- 'native', 'api', 'link', 'iframe'
  created_at TIMESTAMP
);

CREATE TABLE workflow_permissions (
  id SERIAL PRIMARY KEY,
  workflow_id INT REFERENCES workflows,
  persona VARCHAR(50),
  can_view BOOLEAN,
  can_create BOOLEAN,
  can_edit BOOLEAN,
  can_delete BOOLEAN,
  can_share BOOLEAN,
  created_at TIMESTAMP
);
```
